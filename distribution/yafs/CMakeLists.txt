cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
project(CopyUSBYafs LANGUAGES C CXX)
file(TO_CMAKE_PATH "${YAFS_SOURCE_DIR}" YAFS_SOURCE_DIR)
file(TO_CMAKE_PATH "${XERCES_SOURCE_DIR}" XERCES_SOURCE_DIR)
if(NOT MSVC)
  message(FATAL_ERROR "This build requires the Microsoft C++ compiler.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(network OFF CACHE BOOL "" FORCE)
set(transcoder windows CACHE STRING "" FORCE)
set(message-loader inmemory CACHE STRING "" FORCE)
set(mfc-debug OFF CACHE BOOL "" FORCE)
add_subdirectory("${XERCES_SOURCE_DIR}" "xerces" EXCLUDE_FROM_ALL)
set(YAFS_SOURCES command_line_parser.cpp fat_device.cpp fat_elements.cpp file_io.cpp
    main.cpp unicode.cpp utils.cpp version.cpp xercesc.cpp)
list(TRANSFORM YAFS_SOURCES PREPEND "${YAFS_SOURCE_DIR}/")
add_executable(yafs ${YAFS_SOURCES})
target_compile_features(yafs PRIVATE cxx_std_14)
target_compile_definitions(yafs PRIVATE WIN_SYSTEM XERCES_STATIC_LIBRARY YAFS_VERSION=${YAFS_VERSION})
target_compile_options(yafs PRIVATE /W3 /utf-8)
target_link_libraries(yafs PRIVATE xerces-c)
add_executable(yafs-xml-smoke xml-smoke.cpp "${YAFS_SOURCE_DIR}/xercesc.cpp")
target_include_directories(yafs-xml-smoke PRIVATE "${YAFS_SOURCE_DIR}")
target_compile_definitions(yafs-xml-smoke PRIVATE WIN_SYSTEM XERCES_STATIC_LIBRARY)
target_compile_options(yafs-xml-smoke PRIVATE /utf-8)
target_link_libraries(yafs-xml-smoke PRIVATE xerces-c)
